cmake_minimum_required(VERSION 3.22)
project(QtLiteWayland VERSION 6.11.2 LANGUAGES C CXX ASM)
find_package(Qt6 REQUIRED COMPONENTS BuildInternals Core CorePrivate Gui GuiPrivate WaylandScannerTools)
qt_build_repo_begin()
find_package(Wayland REQUIRED COMPONENTS Client Cursor)
find_package(XKB REQUIRED)
find_package(EGL REQUIRED)
set(QT_FEATURE_wayland ON)
set(QT_FEATURE_waylandscanner ON)
# Keep the installed platform and integration plugins; rebuild only their library.
set(QT_FEATURE_wayland_client OFF)
set(QT_FEATURE_wayland_datadevice ON)
set(QT_FEATURE_wayland_client_primary_selection ON)
add_subdirectory(src/plugins/platforms/wayland)
qt_build_repo_end()
